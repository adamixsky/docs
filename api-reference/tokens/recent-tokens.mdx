## Overview

Retrieve a list of tokens that have been recently called by channels, sorted by the most recent call date.

## Request

<ParamField query="page" type="integer" default="1">
  Page number (starting from 1)
</ParamField>

<ParamField query="limit" type="integer" default="10">
  Number of results per page
</ParamField>

## Response

<ResponseField name="data" type="array">
  Array of recent token objects.
  
  <Expandable title="Recent Token Object">
    <ResponseField name="address" type="string">
      Token contract address
    </ResponseField>
    
    <ResponseField name="name" type="string">
      Token name
    </ResponseField>
    
    <ResponseField name="symbol" type="string">
      Token symbol
    </ResponseField>
    
    <ResponseField name="logo_url" type="string | null">
      URL to token logo
    </ResponseField>
    
    <ResponseField name="chain" type="string">
      Blockchain network (e.g., "solana", "bsc")
    </ResponseField>
    
    <ResponseField name="current_fdv" type="number">
      Current fully diluted valuation
    </ResponseField>

    <ResponseField name="ath" type="number">
      All-time high FDV value
    </ResponseField>
    
    <ResponseField name="sufficient_liquidity" type="boolean">
      Whether token has sufficient liquidity
    </ResponseField>

    <ResponseField name="unique_channels" type="integer">
      Number of unique channels that called this token
    </ResponseField>
    
    <ResponseField name="total_subscribers" type="integer">
      Combined subscriber count
    </ResponseField>

    <ResponseField name="most_recent_call" type="string">
      ISO 8601 timestamp of most recent call
    </ResponseField>
    
    <ResponseField name="channel_calls" type="array">
      List of recent calls from different channels.
      
      <Expandable title="Channel Call Object">
        <ResponseField name="id" type="number">Unique call ID</ResponseField>
        <ResponseField name="channel_id" type="string">Calling channel's ID</ResponseField>
        <ResponseField name="channel_name" type="string">Calling channel's name</ResponseField>
        <ResponseField name="subscriber_count" type="number">Channel's subscriber count</ResponseField>
        <ResponseField name="profile_photo_url" type="string">Channel's profile photo</ResponseField>
        <ResponseField name="channel_link" type="string">Link to the channel</ResponseField>
        <ResponseField name="message_id" type="number">ID of the call message</ResponseField>
        <ResponseField name="message_link" type="string">Link to the call message</ResponseField>
        <ResponseField name="fdv_at_call" type="number">FDV at time of call</ResponseField>
        <ResponseField name="highest_fdv" type="number">Highest FDV since call</ResponseField>
        <ResponseField name="lowest_fdv" type="number">Lowest FDV since call</ResponseField>
        <ResponseField name="highest_fdv_at" type="string">Timestamp of highest FDV</ResponseField>
        <ResponseField name="lowest_fdv_at" type="string">Timestamp of lowest FDV</ResponseField>
        <ResponseField name="called_at" type="string">Timestamp of the call</ResponseField>
        <ResponseField name="win_rate" type="number">Channel's win rate</ResponseField>
        <ResponseField name="avg_gain" type="number">Channel's average gain</ResponseField>
        <ResponseField name="platform" type="string">Platform of the channel</ResponseField>
      </Expandable>
    </ResponseField>
    
    <ResponseField name="website" type="string | null">
      Token website URL
    </ResponseField>
    
    <ResponseField name="telegram_link" type="string | null">
      Telegram community link
    </ResponseField>
    
    <ResponseField name="twitter_link" type="string | null">
      Twitter/X account link
    </ResponseField>

    <ResponseField name="last_updated" type="string">
      ISO 8601 timestamp of last data update
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="page" type="integer">
  Current page number
</ResponseField>

<ResponseField name="limit" type="integer">
  Results per page
</ResponseField>

<ResponseField name="total" type="integer">
  Total number of results
</ResponseField>

<ResponseField name="total_pages" type="integer">
  Total number of pages
</ResponseField>

const params = new URLSearchParams({

## Notes

- Tokens are sorted by the `most_recent_call` date (newest first).
- The `channel_calls` array provides a detailed summary of recent calls from different channels.
- `ath` (All-Time High) represents the highest FDV the token has reached.
