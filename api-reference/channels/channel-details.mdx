---
title: 'Channel Details'
api: 'GET /api/channels/{channelId}/info'
description: 'Get detailed information and statistics for a specific channel'
---

## Overview

Retrieve comprehensive information about a specific channel including detailed statistics for multiple timeframes. The response provides a flat structure with performance metrics suffixed by their respective time periods (e.g., _1d, _7d, _all).

## Request

<ParamField path="channelId" type="string" required>
  Unique channel identifier (e.g., "@ExampleChannel")
</ParamField>

## Response

<ResponseField name="channel_id" type="string">
  Unique identifier for the channel
</ResponseField>

<ResponseField name="username" type="string">
  Channel username
</ResponseField>

<ResponseField name="name" type="string">
  Display name of the channel
</ResponseField>

<ResponseField name="link" type="string">
  URL to the channel
</ResponseField>

<ResponseField name="profile_photo_url" type="string">
  URL to the channel's profile picture
</ResponseField>

<ResponseField name="platform" type="string">
  Platform: "telegram" or "twitter"
</ResponseField>

<ResponseField name="subscriber_count" type="integer">
  Number of subscribers
</ResponseField>

<ResponseField name="calls_analyzed" type="integer">
  Number of calls that have been fully analyzed
</ResponseField>

<ResponseField name="total_calls_1d" type="integer">Total calls in the last 24 hours</ResponseField>
<ResponseField name="total_calls_3d" type="integer">Total calls in the last 3 days</ResponseField>
<ResponseField name="total_calls_7d" type="integer">Total calls in the last 7 days</ResponseField>
<ResponseField name="total_calls_14d" type="integer">Total calls in the last 14 days</ResponseField>
<ResponseField name="total_calls_30d" type="integer">Total calls in the last 30 days</ResponseField>
<ResponseField name="total_calls_all" type="integer">Total calls of all time</ResponseField>

<ResponseField name="win_rate_1d" type="number">Win rate for the last 24 hours</ResponseField>
<ResponseField name="win_rate_3d" type="number">Win rate for the last 3 days</ResponseField>
<ResponseField name="win_rate_7d" type="number">Win rate for the last 7 days</ResponseField>
<ResponseField name="win_rate_14d" type="number">Win rate for the last 14 days</ResponseField>
<ResponseField name="win_rate_30d" type="number">Win rate for the last 30 days</ResponseField>
<ResponseField name="win_rate_all" type="number">Win rate for all time</ResponseField>

<ResponseField name="avg_gain_1d" type="number">Average gain for the last 24 hours</ResponseField>
<ResponseField name="avg_gain_3d" type="number">Average gain for the last 3 days</ResponseField>
<ResponseField name="avg_gain_7d" type="number">Average gain for the last 7 days</ResponseField>
<ResponseField name="avg_gain_14d" type="number">Average gain for the last 14 days</ResponseField>
<ResponseField name="avg_gain_30d" type="number">Average gain for the last 30 days</ResponseField>
<ResponseField name="avg_gain_all" type="number">Average gain for all time</ResponseField>

<ResponseField name="x5_rate_all" type="number">Percentage of calls achieving 5x returns (all time)</ResponseField>
<ResponseField name="x10_rate_all" type="number">Percentage of calls achieving 10x returns (all time)</ResponseField>

<ResponseField name="unique_tokens" type="integer">
  Number of unique tokens called
</ResponseField>

<ResponseField name="last_call_at" type="string">
  ISO 8601 timestamp of the most recent call
</ResponseField>

<ResponseField name="first_call_at" type="string">
  ISO 8601 timestamp of the first call
</ResponseField>

<ResponseField name="intelligent_sl" type="number | null">
  Intelligent stop-loss value
</ResponseField>

<ResponseField name="intelligent_tp" type="number | null">
  Intelligent take-profit value
</ResponseField>

<ResponseField name="profit_risk_ratio" type="number | null">
  Profit to risk ratio
</ResponseField>

<ResponseField name="rds_score" type="number | null">
  RDS (Risk-Diversification-Success) score
</ResponseField>

<ResponseField name="survival_rate_percent" type="number | null">
  Percentage of tokens that maintained value
</ResponseField>

<ResponseField name="time_to_ath_minutes" type="number | null">
  Average time to reach all-time high in minutes
</ResponseField>

<ResponseField name="status" type="string">
  Analysis status of the channel (e.g., "pending")
</ResponseField>

<ResponseField name="updated_at" type="string | null">
  ISO 8601 timestamp of the last data update
</ResponseField>
