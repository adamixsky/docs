## Overview

Search for social media channels that make cryptocurrency token calls. Find channels by name, username, or keywords.

## Request

<ParamField query="q" type="string">
  Search query string. Can be channel name, username, or keywords.
</ParamField>

<ParamField query="limit" type="integer" required>
  Maximum number of results to return.
</ParamField>

## Response

<ResponseField name="channels" type="array">
  Array of channel objects matching the search criteria.
  
  <Expandable title="Channel Object">
    <ResponseField name="channel_id" type="string">
      Unique identifier for the channel (e.g., "@ExampleChannel")
    </ResponseField>
    
    <ResponseField name="username" type="string">
      Channel username
    </ResponseField>
    
    <ResponseField name="name" type="string">
      Display name of the channel
    </ResponseField>
    
    <ResponseField name="link" type="string">
      URL to the channel
    </ResponseField>
    
    <ResponseField name="subscriber_count" type="integer">
      Number of subscribers
    </ResponseField>
    
    <ResponseField name="profile_photo_url" type="string">
      URL to profile picture
    </ResponseField>
    
    <ResponseField name="platform" type="string">
      Platform: "telegram" or "twitter"
    </ResponseField>
    
    <ResponseField name="win_rate" type="string">
      Overall win rate percentage
    </ResponseField>
    
    <ResponseField name="avg_gain" type="string">
      Average gain percentage
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="count" type="integer">
  Total number of results found
</ResponseField>

## Example Request

<CodeGroup>

```bash cURL
curl -X GET "https://prod.api.sauron.outlight.fun/api/channels/search?q=crypto&limit=5"
```

```python Python
import requests

response = requests.get(
    "https://prod.api.sauron.outlight.fun/api/channels/search",
    params={
        "q": "crypto",
        "limit": 5
    }
)

data = response.json()
```

```javascript JavaScript
const response = await fetch(
  "https://prod.api.sauron.outlight.fun/api/channels/search?q=crypto&limit=5"
);

const data = await response.json();
```

```typescript TypeScript
interface SearchChannelsResponse {
  channels: Channel[];
  count: number;
}

const response = await fetch(
  "https://prod.api.sauron.outlight.fun/api/channels/search?q=crypto&limit=5"
);

const data: SearchChannelsResponse = await response.json();
```

</CodeGroup>
