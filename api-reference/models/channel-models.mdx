## Overview

This page documents the data models used for channel-related endpoints in the Outlight API.

## Channel

Basic channel information object returned by `list` and `search` endpoints.

```typescript
interface Channel {
  channel_id: string;
  username: string;
  name: string;
  link: string;
  subscriber_count: number;
  profile_photo_url: string;
  platform: "telegram" | "twitter";
  calls_analyzed: number;
  win_rate: string;
  avg_gain: string;
  total_calls_30d: number;
  win_rate_30d: string;
  avg_gain_30d: string;
  x5_rate: string;
  x10_rate: string;
  last_updated: string;
}
```

## ChannelInfo

Extended channel information returned by `GET /api/channels/{channelId}/info`. This response contains both a flat structure with basic metrics and a nested object with advanced analytics.

```typescript
// Nested object for advanced, calculated analytics
interface TimeframeAnalytics {
  time_to_ath_minutes: number;
  survival_rate_percent: number;
  profit_risk_ratio: number | null;
  average_gain_percent: number;
  intelligent_tp: number;
  intelligent_sl: number;
  rds_score: number;
  channel_style: string;
  calculated_at: string;
  updated_at: string;
  timeframe_days: number;
}

// The main response object for the Channel Details endpoint
interface ChannelInfo {
  // Basic Info
  channel_id: string;
  username: string;
  name: string;
  link: string;
  profile_photo_url: string;
  platform: "telegram" | "twitter";
  subscriber_count: number;
  calls_analyzed: number;

  // Basic Timeframe Metrics (Flat Structure)
  total_calls_1d: number;
  win_rate_1d: number;
  avg_gain_1d: number;
  x5_rate_1d: number;
  x10_rate_1d: number;
  // ... (metrics for 3d, 7d, 14d, 30d follow the same pattern)
  total_calls_all: number;
  win_rate_all: number;
  avg_gain_all: number;
  x5_rate_all: number;
  x10_rate_all: number;

  // Top-Level Advanced Metrics (often null)
  unique_tokens: number;
  last_call_at: string;
  first_call_at: string;
  time_to_ath_minutes: number | null;
  survival_rate_percent: number | null;
  profit_risk_ratio: number | null;
  average_gain_percent: number | null;
  intelligent_tp: number | null;
  intelligent_sl: number | null;
  rds_score: number | null;
  channel_style: string | null;
  status: string;

  // Nested Advanced Analytics by Timeframe
  analytics_by_timeframe: {
    [key: string]: TimeframeAnalytics; // e.g., "30d": { ... }
  };
}
```

## ChannelCall

Represents a specific token call made by a channel, returned by `GET /api/channels/{channelId}/calls`.

```typescript
interface ChannelCall {
  id: string;
  channel_id: string;
  token_address: string;
  token_name: string;
  token_symbol: string;
  token_logo_url: string;
  message_id: string;
  message_link: string;
  message_content: string;
  message_html: string;
  called_at: string;
  fdv_at_call: string;
  current_fdv: string;
  highest_fdv: string;
  lowest_fdv: string;
  highest_fdv_at: string;
  lowest_fdv_at: string;
  current_gain: string;
  highest_gain: string;
  website: string | null;
  telegram_link: string | null;
  twitter_link: string | null;
  unique_channels: string;
  total_subscribers: string;
}
```
