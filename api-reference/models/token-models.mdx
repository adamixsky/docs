## Overview

This page documents the data models used for token-related endpoints in the Outlight API.

## TokenSearchResult

Basic token information returned by the `GET /api/tokens/search` endpoint.

```typescript
interface TokenSearchResult {
  address: string;
  name: string;
  symbol: string;
  logo_url: string | null;
  current_fdv: string;
  sufficient_liquidity: boolean;
  unique_channels: number;
  total_subscribers: number;
  call_count: string;
  last_updated: string;
  website?: string | null;
  telegram_link?: string | null;
  twitter_link?: string | null;
}
```

## DetailedTokenCall

This is a detailed call object used within both `RecentToken` and `MostCalledToken` responses.

```typescript
interface DetailedTokenCall {
  id: number;
  channel_id: string;
  channel_name: string;
  subscriber_count: number;
  profile_photo_url: string;
  channel_link: string;
  message_id: number;
  message_link: string;
  fdv_at_call: number;
  highest_fdv: number;
  lowest_fdv: number;
  highest_fdv_at: string;
  lowest_fdv_at: string;
  called_at: string;
  win_rate: number;
  avg_gain: number;
  platform: string;
}
```

## RecentToken

Detailed token information returned by the `GET /api/tokens/recent` endpoint.

```typescript
interface RecentToken extends TokenSearchResult {
  chain: "solana" | "bsc";
  ath: number;
  most_recent_call: string;
  channel_calls: DetailedTokenCall[]; // Re-uses the detailed model
}
```

## MostCalledToken

Token information ranked by call frequency, returned by `GET /api/tokens/most-called`.

```typescript
// This model has been completely rebuilt to match the API response.
interface MostCalledToken {
  address: string;
  name: string;
  symbol: string;
  logo_url: string | null;
  current_fdv: number;
  website: string | null;
  telegram_link: string | null;
  twitter_link: string | null;
  last_updated: string;
  sufficient_liquidity: boolean;
  chain: string;
  unique_channels: number;
  total_calls: string; // Note: API returns this as a string
  last_call_at: string;
  channel_calls: DetailedTokenCall[]; // Re-uses the detailed model
  total_subscribers: number;
  ath: number;
}
```

## TokenResponse

Complete information for a single token, returned by `GET /api/tokens/{address}`.

```typescript
interface TokenResponse {
  address: string;
  name: string;
  symbol: string;
  logo_url: string | null;
  website: string | null;
  telegram_link: string | null;
  twitter_link: string | null;
  price: number | null;
  current_fdv: number | null;
  ath: number | null;
  volume: number | null;
  sufficient_liquidity: boolean;
  total_holders: number | null;
  top_10_holders_percentage: number | null;
  unique_channels: number;
  total_subscribers: number;
  avg_gain: number | null;
  last_updated: string;
  channel_calls: TokenDetailChannelCall[];
}
```
