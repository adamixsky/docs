## 📃 List Channels with Pagination
<APIPlayground
  method="GET"
  url="https://prod.api.sauron.outlight.fun/api/channels"
  queryParameters={{
    page: "number (optional)",
    limit: "number (optional)",
    timeframe: "string (optional)",
    sort: "string (optional)",
    order: "asc|desc (optional)"
  }}
  responseExample={{
    "channels": ["Channel"],
    "totalChannels": 0,
    "totalPages": 0,
    "currentPage": 1

/>
